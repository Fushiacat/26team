### 2. Тесты БД

Файл: `tests/test_db.py`

```python
import pytest
from src.db import create_user, get_user, update_user, get_all_users, get_users_by_role

@pytest.mark.asyncio
async def test_create_user():
    user = await create_user(
        telegram_id=123456,
        name="Тест",
        direction="09.03.01",
        role="seeker",
        description="Тестовое описание"
    )
    assert user is not None
    assert user["telegram_id"] == 123456
    assert user["name"] == "Тест"

@pytest.mark.asyncio
async def test_get_user():
    await create_user(111, "Алексей", "09.03.01", "seeker", "frontend")
    user = await get_user(111)
    assert user is not None
    assert user["name"] == "Алексей"

@pytest.mark.asyncio
async def test_get_user_not_found():
    user = await get_user(999999)
    assert user is None

@pytest.mark.asyncio
async def test_update_user():
    await create_user(222, "Мария", "09.03.03", "organizer", "backend")
    await update_user(222, name="Мария Иванова")
    user = await get_user(222)
    assert user["name"] == "Мария Иванова"

@pytest.mark.asyncio
async def test_create_user_replace():
    """INSERT OR REPLACE при повторном telegram_id."""
    await create_user(333, "Вася", "09.03.01", "seeker", "first")
    await create_user(333, "Вася2", "09.03.03", "organizer", "second")
    user = await get_user(333)
    assert user["name"] == "Вася2"
    assert user["role"] == "organizer"

@pytest.mark.asyncio
async def test_get_users_by_role():
    await create_user(441, "A", "09.03.01", "seeker", "desc")
    await create_user(442, "B", "09.03.01", "organizer", "desc")
    await create_user(443, "C", "09.03.01", "seeker", "desc")
    seekers = await get_users_by_role("seeker")
    assert len(seekers) == 2
    organizers = await get_users_by_role("organizer")
    assert len(organizers) == 1
```

### 3. Тесты матчинга

Файл: `tests/test_match.py`

```python
import pytest
from src.db import create_user, get_users_by_role

@pytest.mark.asyncio
async def test_match_seeker_finds_organizer():
    """Seeker должен найти organizer'ов."""
    await create_user(501, "Организатор", "09.03.01", "organizer", "team lead")
    await create_user(502, "Искатель", "09.03.03", "seeker", "frontend")
    organizers = await get_users_by_role("organizer")
    assert len(organizers) == 1
    assert organizers[0]["telegram_id"] == 501

@pytest.mark.asyncio
async def test_match_organizer_finds_seeker():
    """Organizer должен найти seeker'ов."""
    await create_user(601, "Искатель", "09.03.01", "seeker", "backend")
    await create_user(602, "Организатор", "09.03.03", "organizer", "idea")
    seekers = await get_users_by_role("seeker")
    assert len(seekers) == 1
    assert seekers[0]["telegram_id"] == 601

@pytest.mark.asyncio
async def test_match_no_duplicates():
    """Не показывать самого себя в результатах (фильтрация на уровне хендлера)."""
    await create_user(701, "Один", "09.03.01", "seeker", "desc")
    all_users = await get_users_by_role("organizer")
    assert len(all_users) == 0
```

### 4. Добавить pytest в зависимости

```toml
dev = [
    "pytest>=8.0",

]
```

Создать `pytest.ini` или добавить в `pyproject.toml`:
```toml

```
